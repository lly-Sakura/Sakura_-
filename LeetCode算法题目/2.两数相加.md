
给你两个非空的链表，表示两个非负的整数。它们每位数字都是按照逆序的方式存储的，并且每个节点只能存储一位数字

请你将两个数相加，并以相同形式返回一个表示和的链表

可以假设除了数字0之外，这两个数都不会以0开头

![525](assets/Pasted%20image%2020261127163147.png)

思路：
因为相加是倒序相加，所以要把链表从头相加，再挨个存储，两数相加大于10要进一位

初始化链表的头节点ret，给ret一个nullptr空指针，while循环，两个链表里的数进行挨个相加sum，相加完后取余%，存储到node里，sum/10=num，再进行三元运算

主要代码：
```
struct ListNode
{
	int val;//指针域
	ListNode* next;//下一个节点
	ListNode() :val(0), next(nullptr) {}
	ListNode(int x) : val(x), next(nullptr) {}
	ListNode(int x, ListNode* next) : val(x), next(next) {}
};

ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
	ListNode* current=0; //存储数据
	ListNode* ret = nullptr; //给ret赋值空指针
	int num = 0; 
	while (l1&&l2)
	{
		int sum = l1->val + l2->val + num; //两个链表的数挨个相加
		ListNode* node = new ListNode(sum % 10); //相加完%10，取余赋值给node
		num = sum / 10; //表示进位，大于10的num会带入到下一个循环进行相加
		ret ? current->next = node : ret = node; //三元运算符判断
		//if (ret!=0)
		//{
		//	current->next = node; //ret不是头节点，current的下一个指针指向node
		//}
		//else {
		//	ret = node; //ret是空指针，那么将node赋值给ret
		//}
		current = node; //node赋值到current里
		l1 = l1->next; //移动到下一个指针
		l2 = l2->next; //移动到下一个指针
	}

	if (l2) //处理两个链表的长度不一
	{
		l1 = l2; //如果l2有剩余，就把剩余部分指向l1
	}

	while (l1) //处理剩下的部分
	{
		int sum = num + l1->val;
		ListNode* node = new ListNode(sum % 10);
		num = sum / 10;
		current->next = node;
		current = node;
		l1 = l1->next;
	}

	if (num)//表示进位判断，num=0，不进位（不执行），num = 常量，执行进位将num链接到current->next里
	{
		ListNode* node = new ListNode(num);
		current->next = node;
	}
	return ret;
}
```

```
//创建链表
ListNode* creatList(int arr[], int size) {
	if (size == 0) return nullptr;
	ListNode* head = new ListNode(arr[0]); //初始化头节点
	ListNode* current = head; //将头节点存储到current里
	for (int i = 1; i < size; i++)
	{
		current->next = new ListNode(arr[i]);
		current = current->next; //移动到下一个指针
	}
	return head; 
}

//打印链表
void printList(ListNode* head) {
	ListNode* current = head;
	while (current != nullptr)
	{	
		cout << current->val;
		if (current->next != nullptr) cout << "->";
		current = current->next;
	}
	cout << endl;
}

int main() {
	int arr1[] = { 2,4,3 };
	ListNode* l1 = creatList(arr1, 3);

	int arr2[] = { 5,6,4 };
	ListNode* l2 = creatList(arr2, 3);

	cout << "链表1:\n";
	printList(l1);
	cout << "链表2:\n";
	printList(l2);

	ListNode* result = addTwoNumbers(l1, l2);
	cout << "相加结果:\n";
	printList(result);

	return 0;
}
```
