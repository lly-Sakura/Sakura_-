经典问题：约瑟夫环
已知n个人（分别用编号1，2，3，……，n表示）围坐在一张圆桌周围，从编号为k的人开始顺时针报数，数到m的那个人出列；他的下一个人又从1开始，还是顺时针报数，数到m的那个人又出列；依次重复下去，直到圆桌上剩余一个人

如下图，假设此时圆桌周围有5个人，要求从编号为3的人开始顺时针数数，数到2的那个人出列

![](assets/Pasted%20image%2020261128143447.png)

出列顺序依次为：

	编号为3的人开始数1，然后4数到2，所以4出列
	4出列后，从5开始数1，1数到2，所以1出列
	1出列后，从2开始数，3数到2，所以3出列
	3出列后，从5开始数，2数到2，所以2出列
	最后只剩下5自己，所以5胜出

完整的代码：
```
#include<stdio.h>
#include<stdlib.h>
//循环链表—约瑟夫问题
typedef struct node {
	int number;
	struct node* next; //节点指针
}Joseph;

//初始化链表，头节点
Joseph* initLink(int n) {
	Joseph* head = (Joseph*)malloc(sizeof(Joseph)); //分配内存
	head->number = 1; //头节点的数据是1
	head->next = NULL; //头节点的指针指向置空
	Joseph* cyyclic = head; //cyyclic游标指针，对链表进行链接

	for (int i = 2; i <= n; i++) 
	{
		Joseph* body = (Joseph*)malloc(sizeof(Joseph)); //分配内存赋值给body
		body->number = i; //body的值跟随i进行逐个赋值
		body->next = NULL; //body的指针置空，因为boby后面还没有指向的值，防止野指针
		cyyclic->next = body; //body的值赋值到cyyclic->next里
		cyyclic = cyyclic->next;//存储好的值指向的指针赋值给cyyclic等待下一个body
	}
	cyyclic->next = head; //首尾相连 形成循环链表
	return head; //返回链表
}

void findAndKillk(Joseph* head, int k, int m) { //初始链表，开始个数，报出的数字
	Joseph* tail = head; 
	//找到链表第一个节点的上一个节点，为删除操作做准备
	while (tail->next != head)//如果tail->next指向的不是头节点
	{
		tail = tail->next;//tail会继续移动到下一个
	}
	Joseph* p = head; //将链表赋给p，去寻找编号k值
	//找到编号为k的人
	while (p->number != k) //p的值不等于k值
	{
		tail = p;//记录当前p的位置给tail
		p = p->next;//p移动到下一个值继续循环
	}
	//从编号为k的人开始，只有符合p->next == p时，说明链表中除了p节点，所有编号都出列
	while (p->next != p) //判断循环链表里是否只剩下最后一个值，
	{
		//找到从p报数1开始，报m的人，并且还要知道（m-1）的人的位置tail，方便做删除操作
		for (int i = 1; i < m; i++) //m是报出的数字
		{
			tail = p; //记录当前p的位置给tail
			p = p->next; //p继续移动到下一个值
		}
		//删除报出m的节点
		//tail(m-1)位置 ->next 指向 p->next（向前移动的位置）
		tail->next = p->next; //从链表上将p节点摘下来
		printf("出列人的编号为%d\n", p->number);
		free(p);
		p = tail->next; //tail->next是p->next，p->next指向的值赋给p继续循环
	}
	
	printf("出列人的编号：%d\n", p->number);
	free(p);
}

int main() {
	printf("输入圆桌上的人数：\n");
	int n;
	scanf_s("%d", &n);
	Joseph* head = initLink(n); //初始化链表，调用initLink()函数
	printf("从第k个人开始报数：\n");
	int k;
	scanf_s("%d", &k);
	printf("数到m的人出列：\n");
	int m;
	scanf_s("%d",&m);
	findAndKillk(head, k, m);
	return 0;
}
```
